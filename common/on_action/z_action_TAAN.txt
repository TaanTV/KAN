adventurer_talent_scouting_on_action = {
	trigger = {
		OR = {
			current_travel_plan ?= { has_travel_option = adventurers_scout_for_talent_option }
			has_perk = toe_the_line_perk
		}
	}
	random_events = {
		chance_of_no_event = {
			if = {
				limit = {
					is_ai = yes
				}
				add = 10
			}
			else = {
				add = 50
			}
		}
		100 = 0
		100 = ep3_travel_option_events.0001
	}
}
